# Size strings

`parse_size(text)` converts a size string to a number of bytes (an `int`).

Format: a number, optional spaces, then a unit.

- Number: one or more digits, optionally followed by a dot and one or more digits (`10`, `1.5`). No sign, no exponent, no underscores, no leading or trailing dot.
- Unit: `B`, `KB`, `MB` or `GB`, case-insensitive. Units are powers of 1024 (`1KB` is 1024 bytes).
- Spaces: any number of ordinary spaces (not other whitespace) may separate the number and the unit. Surrounding whitespace around the whole string is ignored.
- Fractions are rounded down to a whole number of bytes (`0.3KB` is 307).
- Anything else raises `ValueError`.

import re

UNITS = {"B": 1, "KB": 1024, "MB": 1024 ** 2, "GB": 1024 ** 3}
_PATTERN = re.compile(r"^(\d+)(KB|MB|GB|B)$")


def parse_size(text):
    m = _PATTERN.match(text.strip().upper())
    if not m:
        raise ValueError("bad size: %r" % (text,))
    return int(m.group(1)) * UNITS[m.group(2)]

import unittest

from cachekit.sizes import parse_size


class ParseSizeTest(unittest.TestCase):
    def test_integer_units(self):
        self.assertEqual(parse_size("10B"), 10)
        self.assertEqual(parse_size("10KB"), 10 * 1024)
        self.assertEqual(parse_size("5MB"), 5 * 1024 ** 2)
        self.assertEqual(parse_size("2GB"), 2 * 1024 ** 3)

    def test_case_insensitive(self):
        self.assertEqual(parse_size("3kb"), 3 * 1024)

    def test_rejects_garbage(self):
        with self.assertRaises(ValueError):
            parse_size("lots")


if __name__ == "__main__":
    unittest.main()
